# ================================================================ # SALUDDATA STEM # Preparar copia independiente con identidad visual NexaTech Studio # # NO MODIFICA: # app\templates\dashboard.html # # CREA: # app\templates\dashboard_nexatech.html # app\templates\dashboard_nexatech.original.html # # ================================================================ $ErrorActionPreference = "Stop" # ------------------------------------------------ # 1. RUTA DEL PROYECTO # ------------------------------------------------ $ProjectRoot = "C:\Users\Eduardo León\SaludData_STEM" $TemplateDir = Join-Path $ProjectRoot "app\templates" $OriginalDashboard = Join-Path $TemplateDir "dashboard.html" $NewDashboard = Join-Path $TemplateDir "dashboard_nexatech.html" $BackupDashboard = Join-Path $TemplateDir "dashboard_nexatech.original.html" Write-Host "" Write-Host "============================================================" -ForegroundColor Cyan Write-Host " SALUDDATA STEM · COPIA NEXATECH STUDIO" -ForegroundColor Cyan Write-Host "============================================================" -ForegroundColor Cyan Write-Host "" # ------------------------------------------------ # 2. VALIDACIONES # ------------------------------------------------ if (-not (Test-Path $ProjectRoot)) { Write-Host "ERROR: No existe el proyecto:" -ForegroundColor Red Write-Host $ProjectRoot -ForegroundColor Yellow exit 1 } if (-not (Test-Path $TemplateDir)) { Write-Host "ERROR: No existe la carpeta de templates:" -ForegroundColor Red Write-Host $TemplateDir -ForegroundColor Yellow exit 1 } if (-not (Test-Path $OriginalDashboard)) { Write-Host "ERROR: No se encontró dashboard.html" -ForegroundColor Red Write-Host $OriginalDashboard -ForegroundColor Yellow exit 1 } # ------------------------------------------------ # 3. LEER DASHBOARD ORIGINAL COMO UTF-8 # ------------------------------------------------ Write-Host "Leyendo dashboard original..." -ForegroundColor Gray $Utf8NoBom = New-Object System.Text.UTF8Encoding($false) $DashboardContent = [System.IO.File]::ReadAllText( $OriginalDashboard, [System.Text.Encoding]::UTF8 ) if ([string]::IsNullOrWhiteSpace($DashboardContent)) { Write-Host "ERROR: dashboard.html está vacío." -ForegroundColor Red exit 1 } # ------------------------------------------------ # 4. CREAR RESPALDO DE LA COPIA SI YA EXISTE # ------------------------------------------------ if (Test-Path $NewDashboard) { $timestamp = Get-Date -Format "yyyyMMdd_HHmmss" $OldCopy = Join-Path $TemplateDir "dashboard_nexatech_backup_$timestamp.html" Copy-Item $NewDashboard $OldCopy -Force Write-Host "Se creó respaldo de la copia existente:" -ForegroundColor Yellow Write-Host $OldCopy -ForegroundColor DarkYellow } # ------------------------------------------------ # 5. GUARDAR COPIA ORIGINAL DEL DASHBOARD # ------------------------------------------------ [System.IO.File]::WriteAllText( $BackupDashboard, $DashboardContent, $Utf8NoBom ) Write-Host "Respaldo creado:" -ForegroundColor Green Write-Host " $BackupDashboard" -ForegroundColor Gray # ------------------------------------------------ # 6. CSS / IDENTIDAD VISUAL NEXATECH # ------------------------------------------------ $NexaCss = @' '@ # ------------------------------------------------ # 7. HEADER + MENÚ # ------------------------------------------------ $NexaHeader = @'
Inicio Home Análisis
NexaTech SaludData STEM
⌂ Inicio ⌬ Home ◈ Análisis
'@ # ------------------------------------------------ # 8. FOOTER # ------------------------------------------------ $NexaFooter = @'

Impulsamos la transformación digital con sentido humano, innovación y tecnología para el crecimiento empresarial.

Enlaces útiles

Inicio Academy / Lab Análisis NexaTech Servicios Productos

Síguenos

f X in ◎ ▶
© 2023 NexaTech. Todos los derechos reservados. | Agencia de Transformación Digital e Innovación Tecnológica
'@ # ------------------------------------------------ # 9. JAVASCRIPT DEL MENÚ # ------------------------------------------------ $NexaScript = @' '@ # ------------------------------------------------ # 10. INSERTAR CSS DESPUÉS DE # ------------------------------------------------ Write-Host "Construyendo copia independiente..." -ForegroundColor Gray $NewContent = $DashboardContent $headPattern = "(?i)" if ($NewContent -match $headPattern) { $NewContent = [regex]::Replace( $NewContent, $headPattern, "$NexaCss`r`n", 1 ) } else { Write-Host "ADVERTENCIA: No se encontró . Se agregará CSS al inicio." -ForegroundColor Yellow $NewContent = $NexaCss + "`r`n" + $NewContent } # ------------------------------------------------ # 11. INSERTAR HEADER DESPUÉS DE # ------------------------------------------------ $bodyPattern = "(?i)]*)>" if ($NewContent -match $bodyPattern) { $NewContent = [regex]::Replace( $NewContent, $bodyPattern, { param($m) $m.Value + "`r`n" + $NexaHeader }, 1 ) } else { Write-Host "ERROR: No se encontró en dashboard.html." -ForegroundColor Red exit 1 } # ------------------------------------------------ # 12. INSERTAR FOOTER ANTES DE # ------------------------------------------------ $bodyClosePattern = "(?i)" if ($NewContent -match $bodyClosePattern) { $NewContent = [regex]::Replace( $NewContent, $bodyClosePattern, $NexaFooter + "`r`n" + $NexaScript + "`r`n", 1 ) } else { Write-Host "ERROR: No se encontró en dashboard.html." -ForegroundColor Red exit 1 } # ------------------------------------------------ # 13. ASEGURAR FUENTES NEXATECH # ------------------------------------------------ $FontBlock = @' '@ if ( $NewContent -notmatch "fonts.googleapis.com/css2" ) { $NewContent = [regex]::Replace( $NewContent, "(?i)", $FontBlock + "`r`n", 1 ) } # ------------------------------------------------ # 14. GUARDAR NUEVA COPIA EN UTF-8 SIN BOM # ------------------------------------------------ [System.IO.File]::WriteAllText( $NewDashboard, $NewContent, $Utf8NoBom ) # ------------------------------------------------ # 15. VERIFICACIONES # ------------------------------------------------ if (-not (Test-Path $NewDashboard)) { Write-Host "" Write-Host "ERROR: No se pudo crear la nueva plantilla." -ForegroundColor Red exit 1 } $SavedContent = [System.IO.File]::ReadAllText( $NewDashboard, [System.Text.Encoding]::UTF8 ) $Checks = @( @{ Name = "Archivo nuevo" Test = (Test-Path $NewDashboard) }, @{ Name = "Header NexaTech" Test = ($SavedContent -match "sd-nexa-header") }, @{ Name = "Menú móvil" Test = ($SavedContent -match "sd-nexa-mobile-overlay") }, @{ Name = "Footer NexaTech" Test = ($SavedContent -match "sd-nexa-footer") }, @{ Name = "Enlace Academy Lab" Test = ($SavedContent -match "nexatech.org/academy/lab.html") }, @{ Name = "Dashboard original intacto" Test = ($DashboardContent -eq [System.IO.File]::ReadAllText( $OriginalDashboard, [System.Text.Encoding]::UTF8 )) } ) Write-Host "" Write-Host "============================================================" -ForegroundColor Cyan Write-Host " VERIFICACIÓN" -ForegroundColor Cyan Write-Host "============================================================" -ForegroundColor Cyan $AllOk = $true foreach ($Check in $Checks) { if ($Check.Test) { Write-Host (" [OK] " + $Check.Name) -ForegroundColor Green } else { Write-Host (" [ERROR] " + $Check.Name) -ForegroundColor Red $AllOk = $false } } # ------------------------------------------------ # 16. INFORMACIÓN FINAL # ------------------------------------------------ Write-Host "" if ($AllOk) { Write-Host "============================================================" -ForegroundColor Green Write-Host " COPIA CREADA CORRECTAMENTE" -ForegroundColor Green Write-Host "============================================================" -ForegroundColor Green Write-Host "" Write-Host "Nueva plantilla:" -ForegroundColor White Write-Host " $NewDashboard" -ForegroundColor Cyan Write-Host "" Write-Host "Respaldo de la copia:" -ForegroundColor White Write-Host " $BackupDashboard" -ForegroundColor Gray Write-Host "" Write-Host "IMPORTANTE:" -ForegroundColor Yellow Write-Host " dashboard.html NO fue modificado." -ForegroundColor Yellow Write-Host "" Write-Host "La nueva copia contiene:" -ForegroundColor White Write-Host " - Identidad visual NexaTech Studio" -ForegroundColor Gray Write-Host " - Menú Inicio / Home / Análisis" -ForegroundColor Gray Write-Host " - Menú hamburguesa para móvil" -ForegroundColor Gray Write-Host " - Logo NexaTech" -ForegroundColor Gray Write-Host " - Footer NexaTech" -ForegroundColor Gray Write-Host " - Dashboard original conservado" -ForegroundColor Gray Write-Host "" Write-Host "============================================================" -ForegroundColor Green } else { Write-Host "" Write-Host "============================================================" -ForegroundColor Red Write-Host " SE DETECTARON PROBLEMAS EN LA VERIFICACIÓN" -ForegroundColor Red Write-Host "============================================================" -ForegroundColor Red exit 1 } # ------------------------------------------------ # 17. OPCIONAL: MOSTRAR ARCHIVOS CREADOS # ------------------------------------------------ Write-Host "" Write-Host "Archivos relacionados:" -ForegroundColor White Get-ChildItem $TemplateDir -Filter "dashboard_nexatech*" | Select-Object Name, Length, LastWriteTime | Format-Table -AutoSize Write-Host "" Write-Host "Listo." -ForegroundColor Green Write-Host ""
